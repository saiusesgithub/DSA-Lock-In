### 1684. Count the Number of Consistent Strings
Easy

initially thought of tried brute forcing but that would need 3 loops 
so directly did this (as soon as i read - only characters and that too unique characters in allowed string , knew i can directly use array instead of hashmap)

```java
class Solution {
    public int countConsistentStrings(String allowed, String[] words) {
        int count = words.length;
        int[] chars = new int[26];
        for (int i =0;i<allowed.length();i++) {
            chars[allowed.charAt(i) - 'a'] = 1;
        }
        for (int j =0;j<words.length;j++) {
            for (int k =0;k<words[j].length();k++) {
                int index = words[j].charAt(k) - 'a';
                if (chars[index] != 1) {
                    count--;
                    break;
                }
            }
        }
        return count;
    }
}
```

```java
class Solution {
    public int countConsistentStrings(String allowed, String[] words) {
        int count = words.length;
        HashSet<Character> set = new HashSet<>();
        for (int i =0;i<allowed.length();i++) {
            set.add(allowed.charAt(i));
        }

        for (int j =0;j<words.length;j++) {
            for (int k =0;k<words[j].length();k++) {
                char c = words[j].charAt(k);
                
                if (!set.contains(c)) {
                    count--;
                    break;
                }
            }
        }
        return count;
    }
}
```
