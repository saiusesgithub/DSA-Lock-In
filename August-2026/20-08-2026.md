### 2220. Minimum Bit Flips to Convert Number
Easy

```java
class Solution {
    public int minBitFlips(int start, int goal) {
        int count = 0;
        for (int i = 0; i < 32; i++) {
            int flag = 1 << i;
            int startBit = start & flag; // 0 or 1 
            int goalBit = goal & flag; // 0 or 1
            if (goalBit != startBit) {
                start = start ^ flag;
                count++;
            }
        }
        return count;
    }
}
```

```java
class Solution {
    public int minBitFlips(int start, int goal) {
        int xorResult = start ^ goal;
        int count = 0;
        for (int i = 0; i < 32; i++) {
            int flag = 1 << i;
            if ((xorResult & flag) != 0) {
                xorResult = xorResult ^ flag;
                count++;
            }
        }
        return count;
    }
}
```
